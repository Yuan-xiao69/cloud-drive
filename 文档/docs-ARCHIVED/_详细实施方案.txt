## 2.2 数据表详细定义

### 2.2.1 RepoFile（仓库文件）【新建】

```prisma
/// 仓库文件表 - 替代原有的 RepoItem
/// 支持文件夹和文件，路径层级通过 parentId 实现
model RepoFile {
  id          String    @id @default(cuid())
  repoId      String    @map("repo_id")
  parentId    String?   @map("parent_id")  // null 表示根目录
  name        String    @map("file_name")   @db.VarChar(255)
  type        String    @map("file_type")   @db.VarChar(20)  // 'file' | 'folder'
  size        Int       @default(0)         @map("file_size")
  mimeType    String?   @map("mime_type")   @db.VarChar(100)
  content     String?   @db.Text            // 小文件直接存储内容
  storageKey  String?   @map("storage_key") @db.VarChar(500)  // MinIO 路径
  hash        String?   @db.VarChar(64)    // SHA256 用于去重
  depth       Int       @default(0)         @map("folder_depth")
  path        String    @db.VarChar(2000)   // 完整路径如 /src/components/Button.tsx
  sortOrder   Int       @default(0)         @map("sort_order")
  isDeleted   Boolean   @default(false)      @map("is_deleted")
  deletedAt   DateTime? @map("deleted_at")
  deletedBy   String?   @map("deleted_by")
  createdAt   DateTime  @default(now())     @map("created_at")
  updatedAt   DateTime  @updatedAt          @map("updated_at")

  // 关系
  repo        Repository @relation(fields: [repoId], references: [id], onDelete: Cascade)
  parent      RepoFile?  @relation("FileParent", fields: [parentId], references: [id])
  children    RepoFile[]  @relation("FileParent")
  versions    FileVersion[]
  shareLinks  ShareLink[]

  // 索引
  @@index([repoId, parentId])
  @@index([repoId, type])
  @@index([repoId, path])
  @@index([hash])
  @@map("repo_file")
}

/// 文件版本历史
model FileVersion {
  id          String   @id @default(cuid())
  fileId      String   @map("file_id")
  content     String?  @db.Text
  storageKey  String?  @map("storage_key")  @db.VarChar(500)
  size        Int      @default(0)           @map("file_size")
  hash        String   @db.VarChar(64)
  message     String?  @db.Text              // 版本说明
  createdBy   String   @map("created_by")
  createdAt   DateTime @default(now())      @map("created_at")

  file        RepoFile  @relation(fields: [fileId], references: [id], onDelete: Cascade)

  @@index([fileId, createdAt])
  @@map("file_version")
}

/// 分享链接
model ShareLink {
  id          String    @id @default(cuid())
  fileId      String?   @map("file_id")
  folderId    String?   @map("folder_id")
  token       String    @unique  @db.VarChar(64)
  password    String?   @db.VarChar(255)    // bcrypt 加密
  maxViews    Int?      @map("max_views")
  viewCount   Int       @default(0)         @map("view_count")
  expiresAt   DateTime? @map("expires_at")
  createdBy   String    @map("created_by")
  createdAt   DateTime  @default(now())      @map("created_at")

  file        RepoFile? @relation(fields: [fileId], references: [id], onDelete: Cascade)
  folder      RepoFile? @relation(fields: [folderId], references: [id], onDelete: Cascade)

  @@index([token])
  @@map("share_link")
}

/// Fork 申请
model ForkRequest {
  id          String    @id @default(cuid())
  repoId      String    @map("repo_id")
  requesterId String    @map("requester_id")
  message     String?   @db.Text
  status      String    @default("pending")  @db.VarChar(20)
                                                    // pending|approved|rejected|expired
  note        String?   @db.Text              // 审批备注
  processedBy String?   @map("processed_by")
  processedAt DateTime? @map("processed_at")
  expiresAt   DateTime? @map("expires_at")
  createdAt   DateTime  @default(now())        @map("created_at")
  updatedAt   DateTime  @updatedAt             @map("updated_at")

  repo        Repository @relation(fields: [repoId], references: [id], onDelete: Cascade)
  requester   User       @relation(fields: [requesterId], references: [id])

  @@index([repoId, status])
  @@index([requesterId])
  @@map("fork_request")
}
```

## 2.3 数据库约束

```sql
-- 唯一约束
ALTER TABLE repo_file ADD CONSTRAINT uk_repo_file_path UNIQUE (repo_id, path);
ALTER TABLE share_link ADD CONSTRAINT uk_share_token UNIQUE (token);

-- 检查约束
ALTER TABLE repo_file ADD CONSTRAINT chk_file_type CHECK (file_type IN ('file', 'folder'));
ALTER TABLE fork_request ADD CONSTRAINT chk_fork_status CHECK (status IN ('pending', 'approved', 'rejected', 'expired'));
ALTER TABLE notification ADD CONSTRAINT chk_priority CHECK (priority IN ('urgent', 'high', 'normal', 'low'));

-- 外键约束
ALTER TABLE repo_file ADD CONSTRAINT fk_repo_file_repo FOREIGN KEY (repo_id) REFERENCES repository(id) ON DELETE CASCADE;
ALTER TABLE repo_file ADD CONSTRAINT fk_repo_file_parent FOREIGN KEY (parent_id) REFERENCES repo_file(id) ON DELETE CASCADE;
```

## 3.1 API 规范

### 3.1.1 RESTful 规范

```
基础格式: /api/v1/{resource}/{id?}/{sub-resource?}

请求格式:
- Content-Type: application/json
- Authorization: Bearer {token}

响应格式:
{
  "success": boolean,
  "data": any,
  "message": string,
  "errors": array,
  "meta": {
    "page": number,
    "pageSize": number,
    "total": number,
    "totalPages": number
  }
}
```

### 3.1.2 HTTP 状态码

| 状态码 | 含义 | 使用场景 |
|--------|------|---------|
| 200 | OK | 成功获取/更新 |
| 201 | Created | 成功创建 |
| 204 | No Content | 成功删除 |
| 400 | Bad Request | 参数错误 |
| 401 | Unauthorized | 未登录 |
| 403 | Forbidden | 权限不足/被限制 |
| 404 | Not Found | 资源不存在 |
| 409 | Conflict | 资源冲突 |
| 429 | Too Many Requests | 请求过于频繁 |
| 500 | Internal Error | 服务器错误 |

### 3.2.2 创建文件/文件夹

```
POST /api/v1/repos/{owner}/{repo}/files

Request Body:
{
  "parentId": string | null,  // null 表示在根目录创建
  "name": string,              // 文件/文件夹名称
  "type": "file" | "folder",
  "content": string,           // 文件内容（type=file 时必需）
  "message": string            // 创建说明（可选）
}

Response 201:
{
  "success": true,
  "data": {
    "id": "clx789...",
    "name": "test.ts",
    "type": "file",
    "size": 1024,
    "path": "/src/test.ts",
    "createdAt": "2024-01-15T10:30:00Z"

Errors:
400: 文件名不合法 / 文件已存在 / 父文件夹不存在
403: 仓库已锁定 / 用户被限制创建文件
404: 仓库不存在
```

```tsx
  y,

  onClose,
  onRename,
  onDelete,
  onMove,
  onCopy,
  onPaste,
  onDownload,
  onShare,
  onView,

  const menuRef = useRef<HTMLDivElement>(null);
  const [copiedFile, setCopiedFile] = useState<string | null>(null);

  // 点击外部关闭
  useEffect(() => {
    const handleClickOutside = (e: MouseEvent) => {
      if (menuRef.current && !menuRef.current.contains(e.target as Node)) {
        onClose();
      }
    };
    document.addEventListener('mousedown', handleClickOutside);
    return () => document.removeEventListener('mousedown', handleClickOutside);
  }, [onClose]);

  // 菜单位置调整（防止超出屏幕）
  const adjustedPosition = useMemo(() => {
    const menuWidth = 200;
    const menuHeight = file.isDeleted ? 150 : 250;
    return {
      x: x + menuWidth > window.innerWidth ? x - menuWidth : x,
      y: y + menuHeight > window.innerHeight ? y - menuHeight : y,
    };
  }, [x, y]);

  const menuItems = [
    { label: '打开', icon: <FolderOpenIcon />, action: onView, shortcut: 'Enter' },
    { label: '重命名', icon: <EditIcon />, action: onRename, shortcut: 'F2' },
    { type: 'divider' as const },
    { label: '复制', icon: <CopyIcon />, action: () => { onCopy(); onClose(); }, shortcut: 'Ctrl+C' },
    { label: '剪切', icon: <CutIcon />, action: onMove, shortcut: 'Ctrl+X' },
    { label: '粘贴', icon: <PasteIcon />, action: onPaste, shortcut: 'Ctrl+V', disabled: !copiedFile },
    { type: 'divider' as const },
    { label: '下载', icon: <DownloadIcon />, action: onDownload, shortcut: 'Ctrl+S' },
    { label: '分享', icon: <ShareIcon />, action: onShare },
    { type: 'divider' as const },
    { label: '删除', icon: <TrashIcon />, action: onDelete, shortcut: 'Del', danger: true },
  ];

  if (file.isDeleted) {
    menuItems.unshift(
      { label: '恢复', icon: <RestoreIcon />, action: onRestore || (() => {}), shortcut: 'R' }
    );
    // 删除已删除文件的菜单
    const deleteIndex = menuItems.findIndex(item => item.label === '删除');
    if (deleteIndex > -1) {
      menuItems[deleteIndex] = { label: '永久删除', icon: <DeleteForeverIcon />, action: onDelete, danger: true };
    }
  }

  return (
    <div
      ref={menuRef}
      className="context-menu"
      style={{ left: adjustedPosition.x, top: adjustedPosition.y }}
    >
      {menuItems.map((item, index) =>
        item.type === 'divider' ? (
```

```tsx
    isDragging,
  } = useSortable({ id: file.id });

function DraggableFileList({
  files,
  onDragEnd,
}: {
  files: FileItem[];
  onDragEnd: (oldIndex: number, newIndex: number) => void;
}) {
  const sensors = useSensors(
    useSensor(PointerSensor),
    useSensor(KeyboardSensor, {
      coordinateGetter: sortableKeyboardCoordinates,
    })
  );

  return (
    <DndContext
      sensors={sensors}
      collisionDetection={closestCenter}
      onDragEnd={(event: DragEndEvent) => {
        const { active, over } = event;
        if (over && active.id !== over.id) {
          const oldIndex = files.findIndex((f) => f.id === active.id);
          const newIndex = files.findIndex((f) => f.id === over.id);
          onDragEnd(oldIndex, newIndex);
        }
      }}
    >
      <SortableContext items={files.map(f => f.id)} strategy={verticalListSortingStrategy}>
        {files.map((file) => (
          <SortableFileItem key={file.id} file={file} />
        ))}
      </SortableContext>
    </DndContext>
  );
}
```

```tsx
      return (
        <CodePreview
          code={content || ''}
          language={getLanguageFromExt(ext)}

    // 视频预览
    if (['mp4', 'webm', 'ogg'].includes(ext || '')) {
      return <VideoPreview src={content} />;
    }

    // 音频预览
    if (['mp3', 'wav', 'ogg'].includes(ext || '')) {
      return <AudioPreview src={content} />;

    return <PlainTextPreview content={content || ''} />;
```

```tsx
// components/layout/Breadcrumb.tsx
interface BreadcrumbItem {
  label: string;
  href?: string;
  icon?: ReactNode;
}

interface BreadcrumbProps {
  items: BreadcrumbItem[];
  maxItems?: number;  // 最多显示的数量，两端省略
  separator?: string;
  onClick?: (item: BreadcrumbItem, index: number) => void;
}

function Breadcrumb({
  items,
  maxItems = 5,
  separator = '/',
  onClick,
}: BreadcrumbProps) {
  const truncatedItems = useMemo(() => {
    if (items.length <= maxItems) return items;

    const first = items.slice(0, 1);
    const last = items.slice(-(maxItems - 1));
    return [...first, { label: '...' }, ...last];
  }, [items, maxItems]);

  return (
    <nav className="breadcrumb" aria-label="面包屑导航">
      {truncatedItems.map((item, index) => (
        <span key={index} className="breadcrumb-item">
          {index > 0 && <span className="separator">{separator}</span>}
          {item.href ? (
            <Link href={item.href} className="breadcrumb-link">
              {item.icon && <span className="icon">{item.icon}</span>}
              <span className="label">{truncate(item.label, 30)}</span>
            </Link>
          ) : (
            <span className="breadcrumb-current">
              {item.icon && <span className="icon">{item.icon}</span>}
              <span className="label">{truncate(item.label, 30)}</span>
            </span>
          )}
        </span>
      ))}
    </nav>
  );
}

// 使用示例
function RepoFilePage() {

      label: part,
      href: `/${owner}/${repoName}/tree/main/${pathParts.slice(2, i + 3).join('/')}`,
```

interface Appeal {
  id: string;
  userId: string;
  userUsername: string;
  userEmail: string;
  appealType: 'penalty' | 'ban' | 'hidden';
  targetId: string;
  content: string;
  status: 'pending' | 'approved' | 'rejected';
  reviewNote: string | null;
  reviewerUsername: string | null;
  createdAt: string;
  reviewedAt: string | null;
  penalty?: Penalty;
}

function AdminAppealsTab({ userRole }: { userRole: string }) {
  const [appeals, setAppeals] = useState<Appeal[]>([]);
  const [showDetailModal, setShowDetailModal] = useState(false);
  const [currentAppeal, setCurrentAppeal] = useState<Appeal | null>(null);
  const [reviewNote, setReviewNote] = useState('');
  const [processing, setProcessing] = useState(false);
  
  // 处理申诉通过 - 解除处罚
  const handleApprove = async () => {
    if (!currentAppeal) return;
    setProcessing(true);
    
    try {
      const res = await api.patch(`/admin/appeals/${currentAppeal.id}`, {
        action: 'approve',
        reviewNote,
      });
      
      if (res.code === 0) {
        fetchAppeals();
        setShowDetailModal(false);
      }
    } finally {
      setProcessing(false);
    }
  };
  
  // 处理申诉驳回
  const handleReject = async () => {
    if (!currentAppeal || !reviewNote) {
      alert('请填写驳回原因');
      return;
    }
    setProcessing(true);
    
    try {
      const res = await api.patch(`/admin/appeals/${currentAppeal.id}`, {
        action: 'reject',
        reviewNote,
      });
      
      if (res.code === 0) {
        fetchAppeals();
        setShowDetailModal(false);
      }
    } finally {
      setProcessing(false);
    }
  };
  
  return (
    <div className="tab-content">
      {/* 申诉列表表格 */}
      <table className="data-table">
        <thead>
          <tr>
            <th>申诉用户</th>
            <th>申诉类型</th>
            <th>申诉内容</th>
            <th>状态</th>
            <th>申诉时间</th>
            <th>操作</th>
          </tr>
        </thead>
        <tbody>
          {appeals.map(appeal => (
            <tr key={appeal.id}>
              <td>@{appeal.userUsername}</td>
              <td><Badge type={appeal.appealType}>{appeal.appealType}</Badge></td>
              <td className="appeal-content">{appeal.content.slice(0, 50)}...</td>
              <td>
                <Badge type={
                  appeal.status === 'pending' ? 'warning' :
                  appeal.status === 'approved' ? 'success' : 'danger'
                }>
                  {appeal.status === 'pending' ? '待处理' :
                   appeal.status === 'approved' ? '已通过' : '已驳回'}
                </Badge>
              </td>
              <td>{formatDate(appeal.createdAt)}</td>
              <td>
                <button 
                  className="btn btn-ghost btn-xs"
                  onClick={() => {
                    setCurrentAppeal(appeal);
                    setShowDetailModal(true);
                  }}
                >
                  详情
                </button>
              </td>
            </tr>
          ))}
        </tbody>
      </table>
      
      {/* 申诉详情弹窗 */}
      {showDetailModal && currentAppeal && (
        <AppealDetailModal
          appeal={currentAppeal}
          reviewNote={reviewNote}
          onNoteChange={setReviewNote}
          onApprove={handleApprove}
          onReject={handleReject}
          onClose={() => setShowDetailModal(false)}
          processing={processing}
        />
      )}
    </div>
  );
}

## 八、数据库变更

### 新增 UserBlock 表

```prisma
model UserBlock {
  id          String   @id @default(cuid())
  blockerId   String   @map("blocker_id")
  blockedId   String   @map("blocked_id")
  createdAt   DateTime @default(now()) @map("created_at")
  
  blocker     User     @relation("Blocker", fields: [blockerId], references: [id], onDelete: Cascade)
  blocked     User     @relation("Blocked", fields: [blockedId], references: [id], onDelete: Cascade)
  
  @@unique([blockerId, blockedId])
  @@index([blockerId])
  @@index([blockedId])
}
```

执行命令:
```bash
npx prisma migrate dev --name add_user_block_table
```

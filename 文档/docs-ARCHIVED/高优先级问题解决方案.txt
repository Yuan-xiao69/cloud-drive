```tsx
'use client';

import { useState } from 'react';
import ReportModal from '@/components/common/ReportModal';

function FlagIcon({ size = 14 }: { size?: number }) {
  return (
    <svg width={size} height={size} viewBox="0 0 16 16" fill="currentColor">
      <path d="M8 1a.5.5 0 0 1 .5.5v.5h5a.5.5 0 0 1 0 1h-5v.5A1.5 1.5 0 0 1 7 4.5v1A1.5 1.5 0 0 1 8.5 7h2a.5.5 0 0 1 0 1h-2A2.5 2.5 0 0 1 6 5.5v-1A2.5 2.5 0 0 1 7.5 2H8V1.5A.5.5 0 0 1 8 1z" />
    </svg>
  );
}

export default function IssueReportButton({
  targetType: initialType,
  targetId,
  targetTitle,
}: {
  targetType: 'issue' | 'comment';
  targetId: string;
  targetTitle?: string;
}) {
  const [showReportModal, setShowReportModal] = useState(false);

  return (
    <>
      <button
        className="btn btn-outline"
        onClick={() => setShowReportModal(true)}
        title="举报"
        style={{
          display: 'inline-flex',
          alignItems: 'center',
          gap: 4,
          padding: '4px 12px',
          fontSize: 13,
          borderColor: 'var(--color-border-default)',
          color: 'var(--color-fg-muted)',
        }}
      >
        <FlagIcon size={14} />
        举报
      </button>
      {showReportModal && (
        <ReportModal
          targetType={initialType}
          targetId={targetId}
          targetTitle={targetTitle}
          onClose={() => setShowReportModal(false)}
        />
      )}
    </>
  );
}
```

【中优先级问题 2-3】API 路由实现
--------------------------------------------------------------------------------

需要新建的 API 路由：

1. src/app/api/v1/me/reports/route.ts - 用户举报列表

```typescript
export async function GET(req: NextRequest): Promise<Response> {
  const authResult = await requireAuth(req);
  if (isAuthError(authResult)) return authResult.error;

  const { user } = authResult;
  const { searchParams } = new URL(req.url);
  const page = parseInt(searchParams.get('page') || '1');
  const limit = parseInt(searchParams.get('limit') || '20');

  const [reports, total] = await Promise.all([
    prisma.report.findMany({
      where: { reporterId: user.id },
      orderBy: { createdAt: 'desc' },
      skip: (page - 1) * limit,
      take: limit,
    }),
    prisma.report.count({ where: { reporterId: user.id } }),
  ]);

  return paginated(reports, { page, limit, total });
}
```

2. src/app/api/v1/me/blocks/route.ts - 用户拉黑列表

```typescript
export async function GET(req: NextRequest): Promise<Response> {
  const authResult = await requireAuth(req);
  if (isAuthError(authResult)) return authResult.error;

  const { user } = authResult;
  const { searchParams } = new URL(req.url);
  const page = parseInt(searchParams.get('page') || '1');
  const limit = parseInt(searchParams.get('limit') || '20');

  const [blocks, total] = await Promise.all([
    prisma.userBlock.findMany({
      where: { blockerId: user.id },
      include: { blocked: { select: { id: true, username: true, avatarUrl: true } } },
      orderBy: { createdAt: 'desc' },
      skip: (page - 1) * limit,
      take: limit,
    }),
    prisma.userBlock.count({ where: { blockerId: user.id } }),
  ]);

  return paginated(blocks, { page, limit, total });
}
```

================================================================================
                           第五部分：设计规范
================================================================================

【举报入口视觉规范】

1. 按钮样式（参考 GitHub）：
   - 边框按钮风格（btn-outline）
   - 颜色：灰色系（不抢占主视觉）
   - 悬停：背景变淡灰色
   - 字体：13px
   - 内边距：4px 12px

2. 位置规范：
   - 讨论/Issue/PR：标题右侧，与「编辑」按钮平级
   - 评论：评论右上角，或「回复」按钮旁
   - 仓库：统计栏最后，文字链接形式

3. 显示条件：
   - 必须登录
   - 不能举报自己的内容
   - 举报按钮使用图标+文字「举报」

【举报 Modal 交互规范】

1. 举报原因选项（已有）：
   - 垃圾信息
   - 辱骂骚扰
   - 不当内容
   - 侵权内容
   - 欺诈钓鱼
   - 其他原因

2. 提交后：
   - 显示成功提示
   - 1.5秒后自动关闭
   - Toast 提示（可选）
